; ====================================================================
; SECTION 1: MPAGE — COMPONENT + FILTER QUERIES (BR_DATAMART_*)
; ====================================================================
; Core tables:
;   br_datamart_category  — MPage definition (category_name, category_mean, layout_flag)
;   br_datamart_report    — Component (report_name, report_mean, br_datamart_category_id)
;   br_datamart_report_filter_r — Component-to-filter relationship
;   br_datamart_filter    — Filter definition (filter_display, filter_mean, filter_category_mean)
;   br_datamart_value     — Filter value (freetext_desc, mpage_param_mean, mpage_param_value,
;                           parent_entity_name, parent_entity_id)
;   br_datamart_flex      — Position/location flex override (parent_entity_id = position code_value)
; layout_flag: 0=Summary, 1=Workflow, 3/4=other
; category_type_flag = 1 for MPages


; ---- 1A. GENERIC: All components + all filters for one MPage (Traditional CCL, full value resolution) ----
; Returns every component, filter, and decoded value for a given MPage.
; Most comprehensive pattern — resolves all known parent_entity_name types.
; Parameters to change: bdc.category_mean (MPage identifier), b.report_name (component), bdf.filter_display (filter)

select
  mpage = bdc.category_name
  , component = b.report_name
  , filter = bdf.filter_display
  , flex_position = evaluate2(
      if(nullind(f.br_datamart_flex_id) = 0)
        if(f.parent_entity_type_flag = 1)
         uar_get_code_display(f.parent_entity_id)
        elseif(f.parent_entity_type_flag = 2)
         uar_get_code_display(f_parent.parent_entity_id)
        endif
      else
         "Default"
      endif
    )
  , flex_location = evaluate2(
      if(nullind(f_parent.br_datamart_flex_id) = 0)
        uar_get_code_display(f.parent_entity_id)
      elseif(f.parent_entity_type_flag = 1)
       "All Facilities"
      else
       "Default"
      endif
    )
  , combined_filter_value = trim(build2((evaluate2(
      if(nullcheck(v.freetext_desc,"TRUE",NULLIND(v.freetext_desc)) > " ")
        if(v.parent_entity_id = 0.00 and v.freetext_desc = "1")
          "Yes"
        elseif(v.parent_entity_id = 0.00 and v.freetext_desc = "0")
          "No"
        elseif(v.parent_entity_name = "CODE_VALUE")
          if(v.parent_entity_id = vcv.code_value and vcv.code_set = 93)
           trim(ves.event_set_cd_disp)
          else
           trim(v.freetext_desc)
          endif
        else
          trim(v.freetext_desc)
        endif
      elseif(cnvtupper(v.mpage_param_mean) in ("MP_LOOK_BACK_UNITS","MP_LOOK_BACK_CUR_ENC"))
        if(nullcheck(v.mpage_param_value,"TRUE",NULLIND(v.mpage_param_value)) > " ")
          if(v.value_type_flag = 1)
            build2("All encounters - Specified time period: ",(trim(v.mpage_param_value)))
          elseif(v.value_type_flag = 2)
            build2("Current encounter only - Specified time period: ",(trim(v.mpage_param_value)))
          endif
        else
          if(v.value_type_flag = 1)
            "All encounters"
          elseif(v.value_type_flag = 2)
            "Current encounter only"
          endif
        endif
      elseif(nullcheck(v.mpage_param_value,"TRUE",NULLIND(v.mpage_param_value)) > " ")
       trim(v.mpage_param_value)
      elseif(v.parent_entity_name = "NOMENCLATURE")
       trim(n.source_string)
      elseif(v.parent_entity_name = "CODE_VALUE")
        if(vcv.code_set = 93)
          trim(ves.event_set_cd_disp)
        else
          trim(vcv.display)
        endif
      elseif(v.parent_entity_name = "DCP_FORMS_REF")
       trim(d.definition)
      elseif(v.parent_entity_name = "ALT_SEL_CAT")
       trim(alc.long_description)
      elseif(v.parent_entity_name = "MLTM_DRUG_CATEGORIES")
       trim(mdc.category_name)
      elseif(v.parent_entity_name = "cr_report_template")
       trim(crt.template_name)
      elseif(v.parent_entity_name = "BR_EVENT_GROUPER")
       trim(breg.grouper_name)
      elseif(v.parent_entity_name = "ORDER_CATALOG_SYNONYM")
       trim(ocs.mnemonic)
      elseif(v.parent_entity_name = "PATHWAY_CATALOG")
       trim(pc.description)
      elseif(v.parent_entity_name = "DMS_CONTENT_TYPE")
       trim(dct.display)
      elseif(v.parent_entity_name = "MLTM_ALR_CATEGORY")
       trim(mac.category_description)
      elseif(v.parent_entity_name = "PRSNL")
       trim(pr.name_full_formatted)
      elseif(v.parent_entity_name = "BR_CCN")
       trim(brc.ccn_name)
      elseif(v.parent_entity_name = "BR_HCO")
       trim(bh.hco_name)
      elseif(v.parent_entity_name = "PERSON")
       trim(per.name_full_formatted)
      elseif(v.parent_entity_name = "WORKING_VIEW")
       trim(wv.display_name)
      elseif(v.parent_entity_name = "HM_EXPECT_SAT")
       trim(hes.expect_sat_name)
      elseif(v.parent_entity_name = "OE_FIELD_MEANING")
       trim(ofm.oe_field_meaning)
      endif))
  ))
from
  br_datamart_category bdc
  , br_datamart_report b
  , br_datamart_report_filter_r r
  , br_datamart_filter bdf
  , br_datamart_value v
  , br_datamart_flex f
  , br_datamart_flex f_parent
  , code_value vcv
  , nomenclature n
  , dcp_forms_ref d
  , alt_sel_cat alc
  , mltm_drug_categories mdc
  , cr_report_template crt
  , br_event_grouper breg
  , order_catalog_synonym ocs
  , pathway_catalog pc
  , dms_content_type dct
  , mltm_alr_category mac
  , prsnl pr
  , br_ccn brc
  , br_hco bh
  , person per
  , working_view wv
  , hm_expect_sat hes
  , oe_field_meaning ofm
  , v500_event_set_code ves
plan bdc
  where bdc.layout_flag in(0,1,3,4)
  and bdc.category_type_flag = 1
  and bdc.category_mean = "VB_MHSOPWFPRIMARYCARE"  ; <-- MPage Identifier
join b
  where b.br_datamart_category_id = bdc.br_datamart_category_id
  ;and b.report_name = "[Component name]"           ; <-- optional component filter
join r
  where r.br_datamart_report_id = b.br_datamart_report_id
join bdf
  where bdf.br_datamart_filter_id = outerjoin(r.br_datamart_filter_id)
  and bdf.filter_display = "Map Catalog Codes"       ; <-- optional filter name
join v
  where v.br_datamart_filter_id = outerjoin(bdf.br_datamart_filter_id)
join f
  where f.br_datamart_flex_id = outerjoin(v.br_datamart_flex_id)
  and f.br_datamart_flex_id > outerjoin(0.00)
join f_parent
  where f_parent.br_datamart_flex_id = outerjoin(f.grouper_flex_id)
  and f_parent.br_datamart_flex_id > outerjoin(0.00)
join vcv
  where vcv.code_value = outerjoin(v.parent_entity_id)
  and vcv.active_ind = outerjoin(1)
join n   where n.nomenclature_id = outerjoin(v.parent_entity_id)
join d   where d.dcp_forms_ref_id = outerjoin(v.parent_entity_id)
join alc where alc.alt_sel_category_id = outerjoin(v.parent_entity_id)
join mdc where mdc.multum_category_id = outerjoin(v.parent_entity_id)
join crt where crt.report_template_id = outerjoin(v.parent_entity_id)
join breg where breg.br_event_grouper_id = outerjoin(v.parent_entity_id)
join ocs where ocs.synonym_id = outerjoin(v.parent_entity_id)
join pc  where pc.pathway_catalog_id = outerjoin(v.parent_entity_id)
join dct where dct.dms_content_type_id = outerjoin(v.parent_entity_id)
join mac where mac.alr_category_id = outerjoin(v.parent_entity_id)
join pr  where pr.person_id = outerjoin(v.parent_entity_id)
join brc where brc.br_ccn_id = outerjoin(v.parent_entity_id)
join bh  where bh.br_hco_id = outerjoin(v.parent_entity_id)
join per where per.person_id = outerjoin(v.parent_entity_id)
join wv  where wv.working_view_id = outerjoin(v.parent_entity_id)
join hes where hes.expect_sat_id = outerjoin(v.parent_entity_id)
join ofm where ofm.oe_field_meaning_id = outerjoin(v.parent_entity_id)
join ves where ves.event_set_cd = outerjoin(v.parent_entity_id)
order by mpage, component, filter, flex_position, flex_location, combined_filter_value
with time = 20


; ---- 1B. GENERIC: Component + Filter, SQL-style LEFT JOIN (good for "Ambulatory Organizer" style) ----
; Same conceptual query using SQL-style syntax. Also resolves value by entity type.
; Parameters: mpage.category_name, r.report_name, f.filter_display

SELECT
  MPage = m.category_name
  , Identifier = m.category_mean
  , Layout = evaluate(m.layout_flag,0,"Summary",1,"Workflow")
  , Position = p.display
  , Component = r.report_name
  , Filter_Name = f.filter_display
  , Value = evaluate2(
      if (v.mpage_param_mean = "mp_*") concat(trim(v.mpage_param_mean),": ",v.mpage_param_value)
      elseif(v.parent_entity_name = "DCP_FORMS_REF") pf.description
      elseif(v.parent_entity_name = "MLTM_DRUG_CATEGORIES") d.category_name
      elseif(v.parent_entity_name = "DMS_CONTENT_TYPE") mg.display
      elseif(v.parent_entity_name = "PATHWAY_CATALOG") pc.description
      elseif(v.parent_entity_name = "NOMENCLATURE") n.mnemonic
      elseif(v.parent_entity_name = "ORDER_CATALOG_SYNONYM") o.mnemonic
      elseif(f.filter_display in ("*look back options","*Look Back Options")
        and v.parent_entity_name = "CODE_VALUE")
        concat(trim(v.freetext_desc)," ",trim(c.display))
      elseif(v.parent_entity_name = "CODE_VALUE") c.display
      else evaluate2(
        if(v.freetext_desc = "1") "Yes"
        elseif(v.freetext_desc = "0") "No"
        else v.freetext_desc
        endif)
      endif)
FROM
  code_value c
  , code_value p
  , ORDER_CATALOG_SYNONYM o
  , ALT_SEL_CAT qf
  , PATHWAY_CATALOG pc
  , DCP_FORMS_REF pf
  , NOMENCLATURE n
  , MLTM_DRUG_CATEGORIES d
  , DMS_CONTENT_TYPE mg
  , br_datamart_flex fl
  , br_datamart_value v
  , br_datamart_filter f
  , BR_DATAMART_REPORT_FILTER_R rf
  , br_datamart_report r
  , br_datamart_category m
plan m
  where m.br_datamart_category_id > 0
  and m.category_mean in ("*")           ; <-- set MPage identifier  (OR use category_name below)
  and m.category_name in ("Ambulatory Organizer")  ; <-- set MPage name
join r
  where r.br_datamart_category_id = m.br_datamart_category_id
  and r.report_name in ("*")             ; <-- optional component filter
join rf where rf.br_datamart_report_id = r.br_datamart_report_id
join f
  where f.br_datamart_filter_id = rf.br_datamart_filter_id
  and f.filter_display in ("*")          ; <-- optional filter name
join v   where v.br_datamart_filter_id = f.br_datamart_filter_id
join fl  where outerjoin(v.br_datamart_flex_id) = fl.br_datamart_flex_id
join p   where outerjoin(fl.parent_entity_id) = p.code_value
join c   where outerjoin(v.parent_entity_id) = c.code_value
join pf  where outerjoin(v.parent_entity_id) = pf.dcp_forms_ref_id
join d   where outerjoin(v.parent_entity_id) = d.multum_category_id
join pc  where outerjoin(v.parent_entity_id) = pc.pathway_catalog_id
join qf  where outerjoin(v.parent_entity_id) = qf.alt_sel_category_id
join n   where outerjoin(v.parent_entity_id) = n.nomenclature_id
join o   where outerjoin(v.parent_entity_id) = o.synonym_id
join mg  where outerjoin(v.parent_entity_id) = mg.dms_content_type_id
order by MPage, Position, r.report_seq, f.filter_seq, v.value_seq
with dontcare = c, maxrec = 50000, time = 30


; ---- 1C. GENERIC: Component + Filter, SQL-style with LEFT JOIN syntax ----
; Variant using explicit LEFT JOIN clauses; most readable for SQL users.
; Parameters: mpage.category_name, comp.br_datamart_category_id filter

SELECT
  mpage = mpage.category_name
  , component = comp.report_name
  , filter = f.filter_display
  , filter_meaning = f.filter_mean
  , filter_category = f.filter_category_mean
  , saved_value =
      IF(v.br_datamart_value_id = 0) "<No saved value>"
      ELSEIF(v.parent_entity_id = 0 AND v.freetext_desc = "0") "No"
      ELSEIF(v.parent_entity_id = 0 AND v.freetext_desc = "1") "Yes"
      ELSEIF(v.parent_entity_name = "CODE_VALUE") UAR_GET_CODE_DISPLAY(v.parent_entity_id)
      ELSEIF(v.parent_entity_name = "DCP_FORMS_REF") dcp.description
      ELSEIF(v.parent_entity_name = "NOMENCLATURE" AND TEXTLEN(TRIM(n.source_string)) = 0)
        n.source_identifier
      ELSEIF(v.parent_entity_name = "NOMENCLATURE" AND NULLIND(n.source_string) = 0)
        BUILD(n.source_identifier, "-", n.source_string)
      ELSEIF(v.parent_entity_name = "PATHWAY_CATALOG") pc.description
      ELSE v.freetext_desc
      ENDIF
  , value_source = v.parent_entity_name
  , value_source_id = v.parent_entity_id
  , flexed_position =
      IF(flex.parent_entity_type_flag = 1 AND flex.grouper_ind = 0)
        UAR_GET_CODE_DISPLAY(flex.parent_entity_id)
      ELSE "<Default Setting>"
      ENDIF
  , last_updated = v.updt_dt_tm "MM/DD/YYYY HH:MM;;q"
  , updated_by =
      IF(pr.position_cd != 0)
        BUILD(pr.name_full_formatted, " (", TRIM(UAR_GET_CODE_DISPLAY(pr.position_cd)), ")")
      ELSE pr.name_full_formatted
      ENDIF
FROM BR_DATAMART_CATEGORY mpage
  , BR_DATAMART_REPORT comp
  , BR_DATAMART_REPORT_FILTER_R f_r
  , BR_DATAMART_FILTER f
  , (LEFT JOIN BR_DATAMART_VALUE v ON v.br_datamart_filter_id = f.br_datamart_filter_id
      AND v.end_effective_dt_tm > SYSDATE)
  , (LEFT JOIN BR_DATAMART_FLEX flex ON flex.br_datamart_flex_id = v.br_datamart_flex_id)
  , (LEFT JOIN PRSNL pr ON pr.person_id = v.updt_id)
  , (LEFT JOIN NOMENCLATURE n ON n.nomenclature_id = v.parent_entity_id
      AND v.parent_entity_name = "NOMENCLATURE")
  , (LEFT JOIN DCP_FORMS_REF dcp ON dcp.dcp_forms_ref_id = v.parent_entity_id
      AND v.parent_entity_name = "DCP_FORMS_REF")
  , (LEFT JOIN PATHWAY_CATALOG pc ON pc.pathway_catalog_id = v.parent_entity_id
      AND v.parent_entity_name = "PATHWAY_CATALOG")
PLAN mpage WHERE 1=1
  AND CNVTUPPER(mpage.category_name) = "AMBULATORY ORGANIZER"  ; <-- MPage name (upper-cased)
  AND mpage.category_type_flag = 1
JOIN comp WHERE comp.br_datamart_category_id = mpage.br_datamart_category_id
JOIN f_r WHERE f_r.br_datamart_report_id = comp.br_datamart_report_id
JOIN f WHERE f.br_datamart_filter_id = f_r.br_datamart_filter_id
JOIN flex JOIN v JOIN pr JOIN n JOIN dcp JOIN pc
ORDER BY comp.mpage_pos_seq, f.filter_seq, flexed_position, v.value_seq
WITH TIME=30, UAR_CODE(D)


; ---- 1D. COMPONENT + FILTER: Specific filter with code_value & event set decoding ----
; Traditional CCL. Good pattern for a specific filter on a specific component.
; Resolves code_value display AND event set names via outerjoin.
; Parameters: c.category_name, r.report_name, f.filter_display

select
mpage=c.category_name
, position_flex=if(v1.br_datamart_flex_id = 0.00)"<default>"
  elseif(v1.br_datamart_flex_id > 0.00)uar_get_code_display(x.parent_entity_id) endif
, component=r.report_name
, filter=f.filter_display
, value=vcv.display
, updt_dt_tm=format(v1.updt_dt_tm,'@SHORTDATETIME')
, updt_by=p.name_full_formatted
from
br_datamart_category c
, br_datamart_report r
, br_datamart_value v
, br_datamart_flex x
, br_datamart_report_filter_r fr
, br_datamart_filter f
, br_datamart_value v1
, code_value vcv
, v500_event_set_code ves
, prsnl p
plan c where c.category_name = "Ambulatory Organizer"  ; <-- MPage name
join r where r.br_datamart_category_id = c.br_datamart_category_id
and r.report_name = "Outstanding Items"                ; <-- component name
join v where v.br_datamart_category_id = c.br_datamart_category_id
and v.parent_entity_id = r.br_datamart_report_id
and v.end_effective_dt_tm > cnvtdatetime(curdate,curtime)
join x where x.br_datamart_flex_id = outerjoin(v.br_datamart_flex_id)
join fr where fr.br_datamart_report_id = v.parent_entity_id
join f where f.br_datamart_category_id = v.br_datamart_category_id
and f.br_datamart_filter_id = fr.br_datamart_filter_id
and f.filter_display = "Provider visit note documented"  ; <-- filter name
join v1 where v1.br_datamart_category_id = f.br_datamart_category_id
and v1.br_datamart_filter_id = f.br_datamart_filter_id
and v1.br_datamart_flex_id = v.br_datamart_flex_id
join vcv where vcv.code_value = outerjoin(v1.parent_entity_id)
and vcv.active_ind = outerjoin(1)
join ves where ves.event_set_cd = outerjoin(v1.parent_entity_id)
join p where p.person_id = outerjoin(v1.updt_id)
order by c.category_name, position_flex, v.value_type_flag, v.value_seq
, f.filter_seq, vcv.display_key


; ---- 1E. COMPONENT + FILTER: Extended value display (label, link, look_back, event sets, forms, drug cats) ----
; Traditional CCL. Handles mp_* mpage_param_mean values and multiple entity types.
; Parameters: c.category_name (MPage name)

select
mpage=c.category_name
, position_flex=if(v1.br_datamart_flex_id = 0.00)"<default>"
  elseif(v1.br_datamart_flex_id > 0.00)uar_get_code_display(x.parent_entity_id) endif
, component=r.report_name
, seq=trim(cnvtstring(v.value_seq))
, filter=f.filter_display
, value=
  if(v1.mpage_param_mean = "mp_look_back_units")
   build2("look_back_units",': ',
   if(v1.value_type_flag = 1)"All encounters"
   elseif(v1.value_type_flag = 2)"Current encounter" endif,
     if(v1.parent_entity_id > 0.00)
     build2(' - ',trim(v1.mpage_param_value),' ',uar_get_code_display(v1.parent_entity_id))endif)
  elseif(v1.mpage_param_mean = "mp_label")
    build2("label_display",': ',trim(v1.mpage_param_value))
  elseif(v1.mpage_param_mean = "mp_link")
    build2("component_link",': ',trim(v1.mpage_param_value))
  elseif(v1.mpage_param_mean = "mp_date_format*")
   build2("date_format",': ',trim(v1.mpage_param_value))
  elseif(v1.mpage_param_mean = "mp_add_label")
   build2("quick_add",': ',trim(v1.mpage_param_value))
  elseif(v1.mpage_param_mean = "mp_exp_collapse")
   build2("exp_collapse",': ',trim(v1.mpage_param_value))
  elseif(v1.mpage_param_mean = "mp_scrolling")
   build2("scrolling",': ',trim(v1.mpage_param_value),' ','rows')
  elseif(v1.parent_entity_name = "CODE_VALUE"
   and vcv.code_set not in (54,93))trim(vcv.display)
  elseif(v1.parent_entity_name = "CODE_VALUE"
  and vcv.code_set = 54)build2(trim(v1.freetext_desc),' - ',trim(vcv.display))
  elseif(v1.parent_entity_name = "CODE_VALUE"
   and vcv.code_set = 93)trim(ves.event_set_name)
  elseif(v1.parent_entity_name = "DCP_FORMS_REF")trim(ref.definition)
  elseif(v1.parent_entity_name = "NOMENCLATURE")trim(n.mnemonic)
  elseif(v1.parent_entity_name = "MLTM_DRUG_CATEGORIES")trim(mdc.category_name)
  else v1.freetext_desc
 endif
from
br_datamart_category c
, br_datamart_report r
, br_datamart_value v
, br_datamart_flex x
, br_datamart_report_filter_r fr
, br_datamart_filter f
, br_datamart_value v1
, code_value vcv
, v500_event_set_code ves
, dcp_forms_ref ref
, nomenclature n
, mltm_drug_categories mdc
plan c where c.category_name = "Inpatient Hospitalist Admit Workflow"  ; <-- MPage name
join r where r.br_datamart_category_id = c.br_datamart_category_id
join v where v.br_datamart_category_id = c.br_datamart_category_id
and v.parent_entity_id = r.br_datamart_report_id
and v.end_effective_dt_tm > cnvtdatetime(curdate,curtime)
join x where x.br_datamart_flex_id = outerjoin(v.br_datamart_flex_id)
join fr where fr.br_datamart_report_id = v.parent_entity_id
join f where f.br_datamart_category_id = v.br_datamart_category_id
and f.br_datamart_filter_id = fr.br_datamart_filter_id
join v1 where v1.br_datamart_category_id = f.br_datamart_category_id
and v1.br_datamart_filter_id = f.br_datamart_filter_id
and v1.br_datamart_flex_id = v.br_datamart_flex_id
join vcv where vcv.code_value = outerjoin(v1.parent_entity_id)
and vcv.active_ind = outerjoin(1)
join ves where ves.event_set_cd = outerjoin(v1.parent_entity_id)
join ref where ref.dcp_forms_ref_id = outerjoin(v1.parent_entity_id)
and ref.active_ind = outerjoin(1)
join n where n.nomenclature_id = outerjoin(v1.parent_entity_id)
and n.active_ind = outerjoin(1)
join mdc where mdc.multum_category_id = outerjoin(v1.parent_entity_id)
order by c.category_name, position_flex, v.value_type_flag, v.value_seq
, f.filter_seq, v1.group_seq, v1.value_seq, value


; ---- 1F. DOCUMENT FILTER AUDIT: Clinical Documentation event-set expansion ----
; Returns Clinical Documentation filter groups and their event codes (expanded from event sets).
; Useful for auditing which note types are in each "Specialty Group" slot.
; Parameters: c.category_name (MPage), r.report_name (component)

select distinct
mpage=c.category_name
, position=uar_get_code_display(x.parent_entity_id)
, component=r.report_name

  , br_datamart_filter f

; ---- 5A. Quick Visit contents (pathway components, order sentences, DTA, note phrases) ----

select
quick_visit=cp.pathway_name, cc.component_display
, detail_1 =
  if(ccd.component_entity_name ="NOMENCLATURE")concat(trim(n.source_string),' (',trim(n.source_identifier),')')
  elseif(ccd.component_entity_name = "NOTE_PHRASE")np.abbreviation
  elseif(ccd.component_entity_name = "ORDER_CATALOG_SYNONYM")ocs2.mnemonic
  elseif(ccd.component_entity_name = "ORDER_SENTENCE")ocs.mnemonic
  elseif(ccd.component_entity_name = "LONG_TEXT_REFERENCE")ltr.long_text
  elseif(ccd.component_entity_name = "CODE_VALUE")cv.display endif
, detail_2 =
  if(os.order_sentence_id > 0)os.order_sentence_display_line
  elseif(ccd.component_entity_name ="NOMENCLATURE")uar_get_code_display(n.source_vocabulary_cd)
  elseif(ccd.component_entity_name = "NOTE_PHRASE")np.abb_description endif
from cp_pathway cp, cp_node cn, cp_component cc, cp_component_detail ccd
, nomenclature n, note_phrase np, order_sentence os, order_catalog_synonym ocs
, order_catalog_synonym ocs2, long_text_reference ltr, code_value cv
plan cp where cp.pathway_type_cd in
(select cv1.code_value from code_value cv1 where cv1.code_set = 4003197 and cv1.cdf_meaning = "QUICKVISIT")
and cp.pathway_name = "AMB Pediatric*"    ; <-- pathway name pattern
and cp.owner_prsnl_id = 0.00
join cn where cn.cp_pathway_id = cp.cp_pathway_id
join cc where cc.cp_node_id = cn.cp_node_id
join ccd where ccd.cp_component_id = cc.cp_component_id
join n where n.nomenclature_id = outerjoin(ccd.component_entity_id)
join np where np.note_phrase_id = outerjoin(ccd.component_entity_id)

join ocs2 where ocs2.synonym_id = outerjoin(ccd.component_entity_id)
join ltr where ltr.long_text_id = outerjoin(ccd.component_entity_id)
join cv where cv.code_value = outerjoin(ccd.component_entity_id)

; ---- 5C. Order activity by position (SQL-style, uses FEDERAL_P0630 schema alias) ----

SELECT oc.description AS orderable
, cat_type.display AS catalog_type, act_type.display AS activity_type, oc.catalog_cd
, pos.display AS order_entry_position
, COUNT(DISTINCT o.order_id) AS nbr_orders
, COUNT(DISTINCT o.person_id) AS nbr_patients
FROM FEDERAL_P0630.ORDERS o
JOIN FEDERAL_P0630.ORDER_CATALOG oc ON oc.catalog_cd = o.catalog_cd
JOIN FEDERAL_P0630.CODE_VALUE cat_type ON cat_type.code_value = oc.catalog_type_cd
JOIN FEDERAL_P0630.CODE_VALUE act_type ON act_type.code_value = oc.activity_type_cd
JOIN FEDERAL_P0630.ORDER_ACTION oa ON oa.order_id = o.order_id
  AND oa.action_type_cd = 2534
JOIN FEDERAL_P0630.PRSNL ap ON ap.person_id = oa.action_personnel_id
JOIN FEDERAL_P0630.CODE_VALUE pos ON pos.code_value = ap.position_cd
  AND pos.display NOT ilike 'VA%'          ; <-- exclude VA positions if needed
WHERE o.orig_order_dt_tm > SYSDATE-90
  AND o.order_status_cd IN (2543, 2546, 2550)
  AND o.template_order_flag IN (0,1,5,7)
  AND o.active_ind = 1
GROUP BY 1, 2, 3, 4, 5 ORDER BY 1, 2, 3, 4, 5
