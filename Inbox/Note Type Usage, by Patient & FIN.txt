select
PATIENT=p.name_full_formatted,
FIN=e.encntr_financial_id,
USER_PERFORMED=pp.name_full_formatted,
RESULT_DATE=ce.event_end_dt_tm,
NOTE_TYPE=uar_get_Code_display(ce.event_cd),
STATUS=uar_get_Code_display(ce.result_status_cd),
ENCOUNTER_ID = ce.encntr_id,
ENCOUNTER_TYPE= uar_get_code_display(e.encntr_type_cd),
MRN = pa.alias,
LOCATION = uar_get_code_display(e.loc_facility_cd)
from
clinical_event ce,
person p,
prsnl pp,
person_alias pa,
encounter e
plan ce
where ce.valid_until_dt_tm > sysdate
and ce.event_end_dt_tm > cnvtdatetime("06-MAY-2022 00:00:00")
and ce.event_end_dt_tm < cnvtdatetime("16-MAY-2022 23:59:59")
;and ce.event_cd = 38713107.00 ; note type event code
join p
where p.person_id = ce.person_id
join pp
where pp.person_id = ce.performed_prsnl_id
join pa
where ce.person_id = pa.person_id
and pa.person_alias_type_cd = (select code_value from code_value where code_set = 4 and cdf_meaning = "MRN")
and pa.end_effective_dt_tm = cnvtdatetime('31-DEC-2100 00:00:00')
join e
where ce.encntr_id = outerjoin(e.encntr_id) 
order by ce.encntr_id
with format (date,";;q"), time=120
