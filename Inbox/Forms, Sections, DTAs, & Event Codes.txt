SELECT
    powerform_name = forms_ref.description
    ,section_name = sec_ref.description
    ,dta_mnemonic = dta.mnemonic
    ,dta_result_type = UAR_GET_CODE_DISPLAY(dta.default_result_type_cd)
    ,event = UAR_GET_CODE_DISPLAY(dta.event_cd)
    ,dta.event_cd

 

FROM DCP_FORMS_REF forms_ref
    ,DCP_FORMS_DEF forms_def
    ,DCP_SECTION_REF sec_ref
    ,DCP_INPUT_REF input_ref
    ,NAME_VALUE_PREFS nvp
    ,DISCRETE_TASK_ASSAY dta

 

PLAN forms_ref WHERE forms_ref.dcp_forms_ref_id = 1103155541 ; 0 -23 Month Well Visit -MHS
    AND forms_ref.active_ind = 1
    AND forms_ref.end_effective_dt_tm > SYSDATE
JOIN forms_def WHERE 1=1 
    AND forms_ref.dcp_form_instance_id = forms_def.dcp_form_instance_id
    ;AND forms_ref.dcp_forms_ref_id = forms_def.dcp_forms_ref_id
    AND forms_def.active_ind = 1
JOIN sec_ref WHERE forms_def.dcp_section_ref_id = sec_ref.dcp_section_ref_id
    AND sec_ref.active_ind = 1
    AND sec_ref.end_effective_dt_tm > SYSDATE
JOIN input_ref WHERE 1=1
    AND sec_ref.dcp_section_instance_id = input_ref.dcp_section_instance_id
    ;AND sec_ref.dcp_section_ref_id = input_ref.dcp_section_ref_id
    AND input_ref.active_ind = 1
JOIN nvp WHERE nvp.parent_entity_id = input_ref.dcp_input_ref_id
    AND nvp.parent_entity_name = "DCP_INPUT_REF"
    AND nvp.merge_name = "DISCRETE_TASK_ASSAY"
JOIN dta WHERE nvp.merge_id = dta.task_assay_cd

 

ORDER BY forms_ref.description, forms_def.section_seq, input_ref.input_ref_seq
WITH TIME=30, UAR_CODE(D), MAXREC=1000
