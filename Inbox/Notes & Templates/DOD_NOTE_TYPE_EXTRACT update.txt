select distinct into $1
OUTBOUND_NOTE_TYPE = nt.note_type_description,            	;Updates existing field
INTERNAL_NOTE_TYPE = uar_get_code_display (nt.event_cd),  	;New field
;event_set = vec.event_set_name,				;ok to remove
;organization = trim(o.org_name),			  	;OK to remove	
loinc_class_code = cvo.alias,
loinc_content_type = cvo1.alias,
loinc_type_code = cvo2.alias

 from
note_type nt,
v500_event_code vec,
v500_event_set_code ves,					
device_output_reltn_r dor,					
organization o,						
code_value cv,
code_value_outbound cvo,
code_value_outbound cvo1,
code_value_outbound cvo2

plan nt
  where nt.data_status_ind = 1
join vec
  where vec.event_cd = nt.event_cd
join ves							
  where ves.event_set_name = vec.event_set_name		
join dor							
  where dor.event_set_cd = ves.event_set_cd			
join o								
  where o.organization_id = dor.organization_id		
  and o.active_ind = 1						
  and o.org_name_key != "ZZ*"
  and o.organization_id = 0				;New parameter					
join cv
  where cv.active_ind = 1
  and cv.code_value = nt.event_cd
join cvo
  where cvo.code_value = outerjoin(cv.code_value)
  and cvo.contributor_source_cd = outerjoin(18024137.00) ;updated outer join with number, but can leave as is in the query
  and cvo.alias_type_meaning = outerjoin("CLASSCODE")

join cvo1
  where cvo1.code_value = outerjoin(cv.code_value)
  and cvo1.contributor_source_cd = outerjoin(18024137.00)
  and cvo1.alias_type_meaning = outerjoin("CONTENTTYPE")
join cvo2
  where cvo2.code_value = outerjoin(cv.code_value)
  and cvo2.contributor_source_cd = outerjoin(18024137.00)
  and cvo2.alias_type_meaning = outerjoin("TYPECODE")

order by nt.note_type_description 			;removed "o.org_name_key," 
with format, separator = " "
