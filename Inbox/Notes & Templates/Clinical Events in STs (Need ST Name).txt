select SMART_TEMPLATE = c.category_name
    , FILTER = f.filter_display
    , EVENT_SET_DISPLAY = uar_get_code_display(cv.code_value)
    , EVENT_SET_CD = cv.code_value
    , FILTER_UPDT_DT_TM = v.updt_dt_tm
    , FILTER_UPDT_PERSON = p1.name_full_formatted
    ;, cv.active_ind, cv.inactive_dt_tm ;<-----------WHEN FILTERING FOR INACTIVED EVENT SET SELECTIONS--|
    ;, CODE_VALUE_UPDT_DT_TM = cv.updt_dt_tm, CODE_VALUE_UPDT_PERSON = p.name_full_formatted ;<---------|
from br_datamart_category c
    , br_datamart_report r
    , br_datamart_report_filter_r rfr
    , br_datamart_filter f
    , br_datamart_value vr
    , br_datamart_value v
    , (left join order_catalog_synonym ocs on (v.parent_entity_id = ocs.synonym_id
        and v.parent_entity_name = "ORDER_CATALOG_SYNONYM"))
    , logical_domain ld
    , code_value cv
    , prsnl p
    , prsnl p1
;br_datamart_category c
    where c.category_mean in (
        "VB_NUTRITIONLABSOP" ;Smart Template Identifier
    )
;br_datamart_report r
    and r.br_datamart_category_id = c.br_datamart_category_id
    and r.report_name in (
        "Clinical Events"
        ,"Reports"
    )
;br_datamart_report_filter_r rfr
    and rfr.br_datamart_report_id = r.br_datamart_report_id
;br_datamart_filter f
    and f.br_datamart_filter_id = rfr.br_datamart_filter_id
    and f.filter_category_mean = "EVENT_SET"
; br_datamart_value vr
    and vr.br_datamart_category_id = c.br_datamart_category_id
    and vr.parent_entity_name = "BR_DATAMART_REPORT"
    and vr.parent_entity_id = r.br_datamart_report_id
    and (vr.mpage_param_mean is NULL OR vr.mpage_param_mean !=  "mp_vb_component_status")
;br_datamart_value v
    and v.br_datamart_filter_id = f.br_datamart_filter_id
    and v.logical_domain_id = vr.logical_domain_id
;logical_domain ld
    and ld.logical_domain_id = v.logical_domain_id
;code_value cv
    and cv.code_value = v.parent_entity_id           
    ;and cv.active_ind = 0 ;<----ONLY SHOW INACTIVE EVENT SETS THAT ARE SELECTED
;prsnl p
    and p.person_id = cv.updt_id
;prsnl p1
    and p1.person_id = v.updt_id
order by f.filter_display
with format(date,";;q")
