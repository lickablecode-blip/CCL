; List all the active forms, regardless of use/activity

 

SELECT 
    forms.dcp_forms_ref_id
    ,forms.dcp_form_instance_id
    ,forms.description
    ,forms.definition
    ,TEXT_RENDITION_EVENT = UAR_GET_CODE_DISPLAY(forms.text_rendition_event_cd)
    ,forms.event_set_name
    ,EVENT = UAR_GET_CODE_DISPLAY(forms.event_cd)
    ,forms.event_cd
    ,forms.width
    ,forms.height
    ,forms.done_charting_ind
    ,forms.enforce_required_ind
    ,forms.flags
    ,FLAG_MEANING = EVALUATE(forms.flags,
        0,"",
        1,"No charting unless all required fields complete",
        2,"Show form in maximized state",
        "")
    
    
FROM DCP_FORMS_REF forms
PLAN forms
WHERE forms.active_ind = 1 AND forms.end_effective_dt_tm > SYSDATE
ORDER BY forms.description

 

WITH TIME=30
