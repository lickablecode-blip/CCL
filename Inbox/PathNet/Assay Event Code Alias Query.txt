SELECT DISTINCT
	O_CATALOG_DISP = UAR_GET_CODE_DISPLAY(O.CATALOG_CD)
	, DTA = UAR_GET_CODE_DISPLAY(P.TASK_ASSAY_CD)
	, EVENT_CODE = UAR_GET_CODE_DISPLAY(CVE.EVENT_CD)
	, C.ALIAS
	, C_CONTRIBUTOR_SOURCE_DISP = UAR_GET_CODE_DISPLAY(C.CONTRIBUTOR_SOURCE_CD)
	, CV1.ACTIVE_DT_TM "@SHORTDATETIMENOSEC"

FROM
	CODE_VALUE_ALIAS   C
	, PROFILE_TASK_R   P
	, ORDER_CATALOG   O
	, CODE_VALUE   CV1
	, CODE_VALUE_EVENT_R   CVE

PLAN CVE
JOIN P WHERE P.TASK_ASSAY_CD = OUTERJOIN(CVE.PARENT_CD)
JOIN O WHERE O.CATALOG_CD = OUTERJOIN(P.CATALOG_CD)      AND O.CATALOG_TYPE_CD = 2513
JOIN C WHERE C.CODE_VALUE = OUTERJOIN(CVE.EVENT_CD)
JOIN CV1 WHERE CV1.CODE_VALUE = CVE.EVENT_CD

ORDER BY
	O_CATALOG_DISP
	, DTA
	, EVENT_CODE
	, C_CONTRIBUTOR_SOURCE_DISP
go
