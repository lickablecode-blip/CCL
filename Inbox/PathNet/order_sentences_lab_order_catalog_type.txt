SELECT 
   O.PRIMARY_MNEMONIC,
   OC_MNEMONIC_TYPE_DISP = UAR_GET_CODE_DISPLAY( OC.MNEMONIC_TYPE_CD ),
   OC.MNEMONIC,
   OCS.ORDER_SENTENCE_DISP_LINE,
   ; OS.ORD_COMMENT_LONG_TEXT_ID,
   OC.ORDER_SENTENCE_ID,
   OS.ORDER_SENTENCE_ID,
   OCS.ORDER_CAT_SENT_R_ID,
   OCS.ORDER_SENTENCE_ID,
   O_CATALOG_TYPE_DISP = UAR_GET_CODE_DISPLAY( O.CATALOG_TYPE_CD ),
   O_CATALOG_DISP = UAR_GET_CODE_DISPLAY( O.CATALOG_CD ),
   OCS.SYNONYM_ID,
   OC_CATALOG_DISP = UAR_GET_CODE_DISPLAY( OC.CATALOG_CD ),
   OC.SYNONYM_ID
FROM
   ORDER_CATALOG  O,
   ORDER_CATALOG_SYNONYM  OC,
   ORDER_SENTENCE  OS,
   ORD_CAT_SENT_R  OCS
PLAN O WHERE  O.CATALOG_TYPE_CD != 2516 
    ;(select code_value from code_value where code_set = 6000 and cdf_meaning = “PHARMACY”)
   and o.catalog_cd > 0
JOIN OC WHERE  OC.CATALOG_CD =  O.CATALOG_CD
JOIN OCS where OC.SYNONYM_ID = OCS.SYNONYM_ID
;JOIN OS WHERE  OS.ORDER_SENTENCE_ID =  OC.ORDER_SENTENCE_ID
JOIN OS WHERE  OS.ORDER_SENTENCE_ID =  OCS.ORDER_SENTENCE_ID
;JOIN  OCS WHERE   OCS.SYNONYM_ID =   OC.SYNONYM_ID
ORDER BY O.PRIMARY_MNEMONIC, OC_MNEMONIC_TYPE_DISP, OC.MNEMONIC, OCS.DISPLAY_SEQ, OCS.ORDER_SENTENCE_DISP_LINE

go
