SELECT Distinct 
ORDER_ACTION=substring(1,15,cv.display), 
;oef.oe_format_id,
OE_FORMAT_NAME=substring(1,80,oef.oe_format_name), 
GROUP_SEQUENCE = oeff.group_seq,
FIELD_SEQUENCE = oeff.field_seq,
;oeff.oe_field_id,
FIELD_NAME=substring(1,50,oefields.description),
FIELD_LABEL = oeff.label_text,
ACCEPT_FLAG = if(oeff.accept_flag=0)"Required"
                elseif(oeff.accept_flag=1)"Optional"
                elseif(oeff.accept_flag=2)"No Display"
                elseif(oeff.accept_flag=3)"Display Only" endif,
DEFAULT=substring(1,15,oeff.default_value), 
VALUE_DEFAULT = cv2.display,
CODE_VALUE = oeff.default_parent_entity_id,
CDL_CHECKED = OEFF.clin_line_ind,
DEPART_CHECKED = oeff.dept_line_ind,
VALUE_REQUIRE = oeff.value_required_ind,
Field_up_tm = oeff.updt_dt_tm,
Field_updt_by = p.name_full_formatted
from order_entry_fields oefields, 
oe_format_fields oeff, 
order_entry_format oef, 
code_value cv,
code_value cv2,
prsnl p
;plan oef where oef.oe_format_name = "Hyperbaric Therapy (KY)"; Search by specific OEF
plan oef where oef.catalog_type_cd =        2513.00 ;pulls all lab OEFs
join oeff where oef.oe_format_id = oeff.oe_format_id 
join cv where oeff.action_type_cd = cv.code_value 
and cv.display = "Order"; order action 
join oefields where oeff.oe_field_id=oefields.oe_field_id 
join cv2 where cv2.code_value = oeff.default_parent_entity_id
join p where p.person_id = outerjoin(oeff.updt_id)
order by oef.oe_format_name, oeff.group_seq, oef.oe_format_id
with time = 60
go
