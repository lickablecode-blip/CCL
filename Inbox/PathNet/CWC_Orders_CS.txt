Select 
 Contract_lab = "",
 Mnemonic = oc.primary_mnemonic,
 TestNumber = oc.dept_Display_Name,
 TestDescription = oc.description,
 SpecimenType = uar_get_code_display(cif.specimen_type_cd),
 container = uar_get_code_display(cif.spec_cntnr_cd),
 Volume = cif.min_vol,
 CPT = "",
 Category = "", 
 IsOrder = "Y",
 IsPanel = "Y",
 OrderAsIndividual = "",
 IsPrivate = "N",
 IsSensitive = "N",
 Panel_Member = oc1.dept_display_name,
 PromptCode = "",
 Keywords = "",
 SpecimenRequirements = "",
 Reference_Values = "",
 Methodology = "",
 TestPerformedAt = "",
 Transport_Temp = "",
 DaysSetUp = "",
 Order_Alone = "",
 PatientPrice = "",
 Activity = UAR_get_code_display(oc.activity_type_cd),
 SubActivity =  UAR_get_code_Display(oc.activity_subtype_cd),
 oc1.primary_mnemonic,
 Panel_Code = oc.catalog_cd,
 Component_Code = ocs.catalog_cd,
 SpecimenCD = cif.specimen_type_cd,
 containerCD = cif.spec_cntnr_cd
 
FROM
 Order_catalog oc,
 collection_info_qualifiers cif,
 Dummyt D1,
 cs_component csc,
 order_catalog_synonym ocs,
 order_catalog oc1
Plan OC where 
  oc.Activity_type_cd in (674,671,696,692) and 
  oc.activity_subtype_cd not in (654476, 2496) and
  oc.orderable_type_flag in (6,8)
Join csc where
 oc.catalog_cd = csc.catalog_cd
 Join ocs where
 csc.comp_id = ocs.synonym_id
Join OC1 where
 ocs.catalog_cd = oc1.catalog_cd
Join D1
Join cif where
 oc.catalog_cd = cif.catalog_cd
 
With outerjoin = D1
