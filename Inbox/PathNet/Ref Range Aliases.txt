create program ARUP_Aliases

prompt 
	Output to FilePrinterMINE = MINE   ; Enter or select the printer or file name to send this report to. 

with OUTDEV



; DVDev DECLARED SUBROUTINES



; DVDev DECLARED VARIABLES



; DVDev Start Coding

select distinct into $OUTDEV
	cv.code_set,
	cv.code_value,
	cv.display,
	cv.description,
	Contributor_Source_IN=uar_get_code_display(cva.contributor_source_cd),
	Inbound_order_alias=cva.alias,
	Contributor_source_OUT=uar_get_code_display(cvo.contributor_source_cd),
	Outbound_order_alias = cvo.alias,
	cv2.code_set,
	cv2.code_value,
	cv2.display,
	cv2.description,
	Contributor_Source_IN=uar_get_code_display(cva2.contributor_source_cd),
	Inbound_result_alias = cva2.alias,
	Contributor_Source_OUT=uar_get_code_display(cvo2.contributor_source_cd),
	Outbound_result_alias = cvo2.alias
 from
	orc_resource_list ol,
	order_catalog oc,
	profile_task_r p,
 	discrete_task_assay d,
	code_value cv,
	code_value cv2,
	code_value_outbound cvo,
	code_value_outbound cvo2,
	code_value_alias cva,
	code_value_alias cva2,
	dummyt d1,
	dummyt d2,
	dummyt d3,
	dummyt d4
 
Plan OL where ol.service_resource_cd in (17583562,17583448)
	and ol.active_ind = 1
Join OC where ol.catalog_cd = oc.catalog_cd
	and oc.active_ind = 1
join cv where cv.code_value = oc.catalog_cd
Join P where ol.catalog_cd = p.catalog_cd and p.active_ind = 1
Join D where d.task_assay_cd = p.task_assay_cd
	and d.active_ind = 1
join cv2 where cv2.code_value = p.task_assay_cd
join d1
join cva where cva.code_value = oc.catalog_cd
	and cva.CONTRIBUTOR_SOURCE_CD = 17628107
join d2
join cvo where cvo.code_value = oc.catalog_cd
	and cvo.CONTRIBUTOR_SOURCE_CD = 17628107
join d3
join cva2 where cva2.code_value = p.task_assay_cd
	and cva2.CONTRIBUTOR_SOURCE_CD = 17628107
join d4
join cvo2 where cvo2.code_value = p.task_assay_cd
	and  cvo2.CONTRIBUTOR_SOURCE_CD = 17628107
 
order by oc.primary_mnemonic, p.sequence
 
with outerjoin=d1, outerjoin=d2, outerjoin=d3, outerjoin=d4, format

end
