; Run both audits listed below and put in the same spreadsheet, one after the next. 
; Next sort by Catalog_CD or Description, then by Parent_ref_id, then by Child_Ref_id. This will put the two audits in the correct sequence.

;This selects all rows where there is a Genlab order as the Parent and a DTA for the Child. These will become the DTA Level Billing Rows.

select
bi.Bill_Item_Id,
Parent_Ref_Id =bi.ext_Parent_reference_id,
Child_Ref_id=bi.ext_Child_reference_id,
oc.Bill_only_ind,
	oc.catalog_cd,
	oc.description,
	oc.primary_mnemonic,
	oc.dept_display_name,
	dta.mnemonic,
	dta.description
	

from
Bill_Item bi,
	order_catalog  oc,
	discrete_task_assay  dta

Plan bi
where bi.active_ind =1

join oc
where 
oc.catalog_cd = bi.ext_parent_reference_id
and oc.activity_type_cd = 692
  and oc.active_ind = 1

join dta
where  dta.task_assay_cd = bi.ext_child_reference_id
  and dta.active_ind = 1

go





;This selects the order rows, where the Child ID = 0, which will become the Order Level Billing Row.

select
bi.Bill_Item_Id,
bi.ext_Parent_reference_id,
bi.ext_Child_reference_id,
oc.Bill_only_ind,
	oc.catalog_cd,
	oc.description,
	oc.primary_mnemonic,
	oc.dept_display_name
from
Bill_Item bi,
	order_catalog oc

Plan bi
where bi.active_ind =1
and bi.ext_child_reference_id =0

join oc
where 
oc.catalog_cd = bi.ext_parent_reference_id
and oc.activity_type_cd = 692
  and oc.active_ind = 1
  
go
