select distinct
	cv.code_set,
	cv.code_value,
	cv.description,
	cv.cki,	
	cv.concept_cki,
	oc.primary_mnemonic,
	oc.dept_display_name,
	cv.collation_seq,



	cv2.code_set,
	cv2.code_value,
	cv2.description,
	cv2.display,
	cv2.cki,
	cv2.concept_cki

	
 from 
	orc_resource_list ol, 
	order_catalog oc, 
	profile_task_r p, 
 	discrete_task_assay d,
	code_value cv, 
	code_value cv2



Plan OL where ol.service_resource_cd in (4260578,4260802 )
	and ol.active_ind = 1
Join OC where ol.catalog_cd = oc.catalog_cd 
	and oc.active_ind = 1
join cv where cv.code_value = oc.catalog_cd
Join P where ol.catalog_cd = p.catalog_cd and p.active_ind = 1
Join D where d.task_assay_cd = p.task_assay_cd 
	and d.active_ind = 1
join cv2 where cv2.code_value = p.task_assay_cd


order by oc.primary_mnemonic, p.sequence
