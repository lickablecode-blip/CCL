***Query for who has PPR access to specific service resource***




Select distinct into mine
    pos = uar_get_code_display(p.position_cd)
    ,p.name_full_formatted
    ,p.username
    ,serv_res= uar_get_code_display(ppr.service_resource_cd)



from
                        prsnl      p
                        ,prsnl_service_resource_reltn        ppr
                        
Plan ppr where ppr.service_resource_cd+0 = ;xxxxx

join p where p.person_id = ppr.prsnl_id
            and p.active_ind =1
            and p.position_cd not in (441,3310549,109868091);dba,dbc Pathnet;dna cerner support
            and p.end_effective_dt_tm  >=cnvtdatetime(curdate,curtime)



order by
            p.position_cd
            ,p.name_last_key
            ,p.username
            ,ppr.service_resource_cd
            ,0

with format, separator = " "

 
go
