SELECT     
dfr.dcp_forms_ref_id, 
dfr.description,
dfr.definition, 
Note_Type = UAR_GET_CODE_DISPLAY(vese3.event_set_cd),
PARENT_EVENT_SET = UAR_GET_CODE_DISPLAY(vese4.event_set_cd)

FROM 
dcp_forms_ref dfr,
v500_event_set_explode vese1,
v500_event_set_explode vese2,
v500_event_set_explode vese3,
v500_event_set_explode vese4,
order_task ot,
dummyt d1,
dummyt d2

Plan dfr
where dfr.active_ind = 1

join vese1
where vese1.event_cd = outerjoin(dfr.event_cd)
and vese1.event_set_level = outerjoin(0)

join vese2
where vese2.event_cd = outerjoin(dfr.event_cd)
and vese2.event_set_level = outerjoin(1)

join vese3
where vese3.event_cd = outerjoin(dfr.text_rendition_event_cd)
and vese3.event_set_level = outerjoin(0)

join vese4
where vese4.event_cd = outerjoin(dfr.text_rendition_event_cd)
and vese4.event_set_level = outerjoin(1)

join d1

join ot
where dfr.dcp_forms_ref_id = ot.dcp_forms_ref_id

join d2

Order by dfr.description
