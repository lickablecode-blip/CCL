; This script pulls retail pharmacy prescriptions, provides info about the originating encounter
; (as opposed to just the lifetime pharmacy encounter linked to the actual order),
; and dispensing information
; David Alt (david.a.alt2.mil@mail.mil, 2020-11-17)

SELECT   
	PRESCRIPTION = UAR_GET_CODE_DISPLAY(o.catalog_cd)
	,SOURCE = UAR_GET_CODE_DISPLAY(o.source_cd)
	,ENTRY_METHOD = EVALUATE2(
		IF(o.prescription_order_id > 0) "PowerChart"
		ELSEIF(od.rxa_prescription_msg_id > 1 OR od.erx_msg_long_text_id > 1) "Med Manager (SureScripts)"
		ELSE "Med Manager (manual entry)"
		ENDIF
	)   
	,FILL_PHARMACY = UAR_GET_CODE_DESCRIPTION(od.rx_nbr_cd)
	,FILL_ENC_TYPE = UAR_GET_CODE_DISPLAY(e.encntr_type_cd)
	,FILL_ENC_FACILITY = UAR_GET_CODE_DISPLAY(e.loc_facility_cd)
	,FILL_ORDER_ID = o.order_id

	,RX_ORDER_ID = o.prescription_order_id
	,RX_ENC_TYPE = UAR_GET_CODE_DISPLAY(rx_e.encntr_type_cd)
	,RX_ENC_FACILITY = UAR_GET_CODE_DISPLAY(rx_e.loc_facility_cd)
	,RX_ENC_NURSE_UNIT = UAR_GET_CODE_DISPLAY(rx_e.loc_nurse_unit_cd)
	,RX_ENC_MEDICAL_SVC = UAR_GET_CODE_DISPLAY(rx_e.med_service_cd)
	
	,ORDERING_PROVIDER = op.name_full_formatted
	
	; not sure exactly how each of these are used, but they aren't in synch
	,RX_ORDER_STATUS = UAR_GET_CODE_DISPLAY(rx_o.dept_status_cd)
	,FILL_ORDER_STATUS = UAR_GET_CODE_DISPLAY(o.dept_status_cd)
	,ORDER_DISPENSE_STATUS = UAR_GET_CODE_DISPLAY(od.dept_status_cd)
	
	,od.display_line
	,FREQUENCY = UAR_GET_CODE_DISPLAY(f.frequency_cd)
	,od.days_supply
	,od.refills_remaining
	,od.total_rx_qty
	,DISPENSE_QTY = od.total_rx_qty - od.qty_remaining ; calculated field
	,od.qty_remaining
	
	,od.order_cost_value
	,od.order_price_value
	,PRICE_CODE = UAR_GET_CODE_DESCRIPTION(od.price_code_cd)
	,DISPENSE_AS_WRITTEN = UAR_GET_CODE_DESCRIPTION(od.daw_cd)
	,HAS_CLAIM = EVALUATE(od.claim_flag,0,"no",1,"yes","")
	;,DRUG_SCHEDULE = UAR_GET_CODE_DESCRIPTION(od.legal_status_cd)
 
 FROM ORDERS o
	,(LEFT JOIN ORDERS rx_o ON o.prescription_order_id = rx_o.order_id AND rx_o.active_ind = 1)
	,(LEFT JOIN ENCOUNTER rx_e ON rx_o.encntr_id = rx_e.encntr_id AND rx_e.active_ind = 1)
	,(LEFT JOIN ORDER_DISPENSE od ON o.order_id = od.order_id)
	,(LEFT JOIN FREQUENCY_SCHEDULE f ON od.frequency_id = f.frequency_id)
	,(LEFT JOIN ORDER_ACTION oa ON o.prescription_order_id = oa.order_id 
		AND oa.action_type_cd = 2534)
	,(LEFT JOIN PRSNL op ON oa.order_provider_id = op.person_id AND op.physician_ind = 1 AND op.active_ind = 1)
	,ENCOUNTER e

PLAN o  
WHERE o.active_ind = 1
	AND o.catalog_type_cd = value(UAR_GET_CODE_BY("DISPLAY",6000,"Pharmacy"))
	;AND o.source_cd = value(UAR_GET_CODE_BY("DISPLAY",6500,"Retail")) ; not sure if this should be included
	AND o.orig_ord_as_flag = 1 ;
	AND o.orig_order_dt_tm >= SYSDATE-60
JOIN e WHERE o.encntr_id = e.encntr_id

; Outer joins go last for better performance
JOIN rx_o
JOIN rx_e
JOIN od
JOIN f
JOIN oa
JOIN op

WITH TIME=30, MAXREC=500 ;remove MAXREC to query all results
