select
mpage=c.category_name
, position_flex=if(v1.br_datamart_flex_id = 0.00)"<default>"
  elseif(v1.br_datamart_flex_id > 0.00)uar_get_code_display(x.parent_entity_id) endif
, component=r.report_name
, updt_dt_tm=format(v.updt_dt_tm,'@SHORTDATETIME')
, updt_by=p.name_full_formatted
from
br_datamart_category c
, br_datamart_report r
, br_datamart_value v
, br_datamart_flex x
, br_datamart_report_filter_r fr
, br_datamart_filter f
, br_datamart_value v1
, prsnl p
plan c where c.category_name = "MHS OPWF*" ; name of MPage
join r where r.br_datamart_category_id = c.br_datamart_category_id
and r.report_name = "Referrals" ; name of component
join v where v.br_datamart_category_id = outerjoin(c.br_datamart_category_id)
and v.parent_entity_id = outerjoin(r.br_datamart_report_id)
and v.end_effective_dt_tm > outerjoin(cnvtdatetime(curdate,curtime))
join x where x.br_datamart_flex_id = outerjoin(v.br_datamart_flex_id)
join fr where fr.br_datamart_report_id = outerjoin(v.parent_entity_id)
join f where f.br_datamart_category_id = outerjoin(v.br_datamart_category_id)
and f.br_datamart_filter_id = outerjoin(fr.br_datamart_filter_id)
;and f.filter_display = 
join v1 where v1.br_datamart_category_id = outerjoin(f.br_datamart_category_id)
and v1.br_datamart_filter_id = outerjoin(f.br_datamart_filter_id)
and v1.br_datamart_flex_id = outerjoin(v.br_datamart_flex_id)
join p where p.person_id = outerjoin(v.updt_id)
order by c.category_name
, position_flex
, v.value_type_flag
, v.value_seq
, f.filter_seq
, v1.group_seq
, v1.value_seq
, v1.br_datamart_value_id
