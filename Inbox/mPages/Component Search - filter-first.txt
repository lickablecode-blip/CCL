/*
* Name:    Mpage Component Query (FAWG)
* Source:  Companion to "Mpage Query (FAWG)" -- component-to-mpage stage only
* Purpose: Find which Mpage(s) a specific component lives on, searching by Component
*          name or meaning. Feed a result's Mpage PK/meaning/name into the Mpage Query
*          to resolve viewpoints and positions for a specific Mpage.
* Version: 20260918
* Notes:
*   TO SEARCH: fill in exactly ONE of the two PLAN COMP filters below (name / meaning).
*/

SELECT
mpage_PK = cat.br_datamart_category_id
, mpage_meaning = cat.category_mean
, mpage = cat.category_name
, mpage_title = val.freetext_desc
, mpage_type =
	EVALUATE(cat.layout_flag,
	0, "0 - Summary Layout",
	1, "1 - Workflow Layout",
	2, "2 - Smart Template",
	3, "3 - Quick Orders and Charges",
	4, "4 - Patient Organizer View",
	5, "5 - Dashboard",
	6, "6 - Organizer View - Provider",
	7, "7 - Emergent Event",
	8, "8 - SCM Purchase Order Organizer",
	9, "9 - SCM Inventory Management Organizer")
, component_PK = comp.br_datamart_report_id
, component_meaning = comp.report_mean
, component = comp.report_name
FROM
br_datamart_report comp ;components - driving table, filtered by search criteria below
, br_datamart_category cat
, br_datamart_report rep
, br_datamart_filter fil
, br_datamart_value val
/***component search -- fill in exactly one of the two lines below***/
PLAN comp
;	WHERE comp.report_name = "Pregnancy Overview*"    ;;SEARCH BY COMPONENT NAME - enter the name, case-sensitive (trailing * ok)
	WHERE comp.report_mean = "MP_VB_WF_PREG_OVERVIEW"     ;;SEARCH BY COMPONENT MEANING - enter the meaning
/***mpage this component lives on***/
JOIN cat
	WHERE cat.br_datamart_category_id = comp.br_datamart_category_id
	AND cat.layout_flag = 1                              ;;restrict to Mpages
	AND NOT cat.category_name IN ("z*", "Z*")
/***mpage title -- see Notes above***/
JOIN rep
	WHERE rep.br_datamart_category_id = OUTERJOIN(cat.br_datamart_category_id)
	AND rep.report_mean = OUTERJOIN("MP_VB_WF_PAGE_LEVEL")
JOIN fil
	WHERE fil.br_datamart_category_id = OUTERJOIN(rep.br_datamart_category_id)
	AND fil.filter_display = OUTERJOIN("View display*")
JOIN val
	WHERE val.br_datamart_category_id = OUTERJOIN(fil.br_datamart_category_id)
	AND val.br_datamart_filter_id = OUTERJOIN(fil.br_datamart_filter_id)
	AND val.br_datamart_flex_id = OUTERJOIN(0.00)        ;;default (non-flexed) value only -- avoids per-position dupes
	AND val.end_effective_dt_tm > OUTERJOIN(cnvtdatetime(curdate,curtime))
ORDER BY
mpage_meaning
WITH maxrec=500, time=30, nocounter
