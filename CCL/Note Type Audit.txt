select
               Note_Type=uar_get_code_display(ec.event_cd)
               , Default_Level=evaluate(nt.default_level_flag,
                                             0,"Encounter",
                                             1,"Person")
               , Publish_Level=evaluate(nt.publish_level,
                                             0,"Default",
                                             1,"In-Progress",
                                             2,"Transcribe",
                                             3,"Unauth",
                                             4,"Auth",
                                             5,"Modify")
               , PrelimFinalBanner=evaluate(nt.banner_ind,
                                             1,"Show",
                                             0,"Hide")
               , Overridable=evaluate(nt.override_level_ind,
                                             1,"Yes",
                                             0,"No")
from note_type nt
               , v500_event_code ec
               , ((select ecode=x.event_cd from v500_event_set_code evs, v500_event_set_explode x
                              where evs.event_set_cd_disp_key = "CLINICALDOC"
                              and evs.event_set_cd = x.event_set_cd) nl)
                              
plan nl
join ec
               where outerjoin(cnvtint(nl.ecode)) = ec.event_cd
join nt
               where outerjoin(ec.event_cd) = nt.event_cd
               and nt.data_status_ind = outerjoin(1)
order by ec.event_cd
