select distinct
application=a.description
, position=c.display
, toc_disp=n.pvc_value
, mpage=cat.category_name
, component=rep.report_name
, fil.filter_display
, param=val.mpage_param_mean
, param_value=val.mpage_param_value
from
view_prefs v
, code_value c
, detail_prefs d
, name_value_prefs n
, application a
, name_value_prefs n2
, name_value_prefs n3
, mp_viewpoint m
, mp_viewpoint_reltn r
, br_datamart_category cat
, br_datamart_report rep
, br_datamart_report_filter_r rf
, br_datamart_filter fil
, br_datamart_value val
, br_datamart_flex x
plan v where v.frame_type = "CHART"
and v.view_name = "DISCERNRPT"
and v.application_number = 600005 ; 600005 = PowerChart
join c where c.code_value = v.position_cd
and c.active_ind = 1
;and c.display != "VA*"
join a where a.application_number = v.application_number
join d where d.application_number = v.application_number
and d.position_cd = v.position_cd
and d.view_name = v.view_name
and d.view_seq = v.view_seq
join n2 where n2.parent_entity_id = v.view_prefs_id
and n2.pvc_name = "DISPLAY_SEQ"
join n3 where n3.parent_entity_id = d.detail_prefs_id
and n3.pvc_name = "REPORT_NAME"
and n3.pvc_value = "<url>$DM_INFO:CONTENT_SERVICE_URL$/mp-content*"
join n where n.parent_entity_id = v.view_prefs_id
and n.pvc_name = "VIEW_CAPTION"
join m where m.viewpoint_name_key = value(substring(findstring('vId="', n3.pvc_value)+5
,(findstring('"&s',trim(n3.pvc_value,7))-(findstring('vId="', n3.pvc_value)+5))
, n3.pvc_value), n3.pvc_value)
join r where r.mp_viewpoint_id = m.mp_viewpoint_id
join cat where cat.br_datamart_category_id = r.br_datamart_category_id
and cat.layout_flag = 1
;and cat.category_name = "OPWF Dentistry" ; name of MPage
join rep where rep.br_datamart_category_id = cat.br_datamart_category_id
join rf where rf.br_datamart_report_id = rep.br_datamart_report_id
join fil where fil.br_datamart_filter_id = rf.br_datamart_filter_id
and fil.br_datamart_category_id = cat.br_datamart_category_id
and fil.filter_display = "Component Layout Options" ; name of filter
join val where val.br_datamart_filter_id = fil.br_datamart_filter_id
and val.br_datamart_category_id = fil.br_datamart_category_id
;and val.mpage_param_mean in ("mp_link","mp_label")
and val.mpage_param_value = "Life-Sustaining Treatment" ; name of component
join x where x.br_datamart_flex_id = outerjoin(val.br_datamart_flex_id)
and x.parent_entity_id = outerjoin(c.code_value)
order by a.description
, cnvtupper(c.display)
, cnvtstring(n2.pvc_value)
, cat.category_name
