	, DD_REF_TEMPLATE   DR
	, CODE_VALUE   CV1

join cv1 where CV1.CODE_VALUE = C.SMART_TEMPLATE_CD
 
ORDER BY
	DR.DESCRIPTION_TXT
 
WITH MAXREC = 5000, NOCOUNTER, SEPARATOR=" ", FORMAT

<!-- DynDoc Usage Query *****(Author, Position, Pt., Encounter Type, FIN, DOS, Title, Status, Note_Type -->

select
    AUTHOR=p.name_full_formatted,
Position=uar_get_Code_display (p.position_cd),
PATIENT=p2.name_full_formatted,
Encounter_Type_Class=uar_get_Code_display (e.encntr_type_class_cd),
Encounter_Type=uar_get_Code_display (e.encntr_type_cd),    
    FIN=ea.alias,
DATE_OF_SERVICE =ce.event_end_dt_tm,
TITLE=ce.event_title_text,
STATUS=uar_get_Code_display(ce.result_status_cd),
NOTE_TYPE=uar_get_Code_display(ce.event_cd)
from
    clinical_event ce,
prsnl p,
person p2,
encntr_alias ea,
encounter e
plan ce where ce.valid_until_dt_tm > sysdate
  and ce.event_end_dt_tm > cnvtdatetime("01-DEC-2020 00:00:00") ;enter start date range for query
     and ce.event_end_dt_tm < cnvtdatetime("11-DEC-2021 23:59:59") ;enter end date range for query    
    and ce.event_class_cd =
    (SELECT CV.CODE_VALUE FROM CODE_VALUE CV WHERE CODE_SET = 53 AND CDF_MEANING in("MDOC", "DOC"))
and ce.entry_mode_cd =
    (SELECT CV.CODE_VALUE FROM CODE_VALUE CV WHERE CODE_SET = 29520 AND CDF_MEANING in ("DYNDOC"))
and ce.parent_event_id = ce.event_id
join p where p.person_id = ce.performed_prsnl_id
join p2 where p2.person_id = ce.person_id
join e
where ce.encntr_id = e.encntr_id
join ea where ea.encntr_id = ce.encntr_id
and ea.encntr_alias_type_cd =
    (SELECT CV.CODE_VALUE FROM CODE_VALUE CV WHERE CODE_SET = 319 AND CDF_MEANING = "FIN NBR")
order by ce.event_end_dt_tm desc
with format (date,";;q")
go

<!-- CCL ST detail query with all details including the long blob -->

SELECT 
dyn_doc_template_name = dd.title_txt 
, dyn_doc_template_desc = dd.description_txt 
, dyn_doc_source = dd.source_txt 
, dyn_doc_xml_blob = l.long_blob 
, dd_st_disp = cv1.display 
, dd_st_def = cv1.definition 
, dd_st_desc = cv1.description 
, dd_st_mean = cv1.cdf_meaning 
, dd_st_cki = cv1.cki 
, dd_st_code_value = cv1.code_value 
, dd_st_object_name = d.object_name 
, dd_st_source = d.source_name 
, dd_st_last_updated_by = d.user_name 
, dd_st_object_group = d.group 
, dd_script_type = if(cnvtupper(d.source_name) = "CER_SCRIPT*") "Canned Script" 
else "Custom Script" 
endif 
FROM 
dd_ref_template dd 
, long_blob_reference l 
, dd_ref_tmplt_cn_tmplt_r drt 
, clinical_note_template cnt 
, code_value cv1 
, dprotect d 
plan dd  
where dd.active_ind = 1 
join l  
where l.long_blob_id = dd.long_blob_ref_id 
join drt 
where drt.dd_ref_template_id = dd.dd_ref_template_id 
join cnt 
where cnt.template_id = drt.clinical_note_template_id 
join cv1 
where cv1.code_value = cnt.smart_template_cd 
and cv1.active_ind = 1 
and cv1.cdf_meaning = "CLINNOTETEMP" 
and cv1.cki > " " 
and cv1.definition != "smart_template_wizard*" 
join d  
where "P" = d.object 
and cnvtupper(cv1.definition) = cnvtupper(d.object_name) 
ORDER BY 
dyn_doc_template_name 
, dd_st_disp 
, dd_st_cki 
WITH nocounter, separator=" ", format
